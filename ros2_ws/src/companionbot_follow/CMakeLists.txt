cmake_minimum_required(VERSION 3.20)
project(companionbot_follow LANGUAGES CXX)
if(DEFINED PYTHON_EXECUTABLE)
  set(Python3_EXECUTABLE "${PYTHON_EXECUTABLE}")
endif()
find_package(ament_cmake REQUIRED)
find_package(ament_cmake_python REQUIRED)
find_package(Python3 REQUIRED COMPONENTS Interpreter Development.Module)
find_package(pybind11 REQUIRED CONFIG)
find_package(rclcpp REQUIRED)
find_package(companionbot_interfaces REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(tf2_msgs REQUIRED)
find_package(nlohmann_json REQUIRED)
find_package(Threads REQUIRED)
# Reuse the exact installed Python Ruckig distribution's C++ development
# artifacts. No vendored sources and no configure-time download/fallback.
execute_process(COMMAND "${Python3_EXECUTABLE}" -c
  "import sysconfig; print(sysconfig.get_path('platlib'))"
  OUTPUT_VARIABLE CB_PYTHON_PLATLIB OUTPUT_STRIP_TRAILING_WHITESPACE
  COMMAND_ERROR_IS_FATAL ANY)
list(PREPEND CMAKE_PREFIX_PATH "${CB_PYTHON_PLATLIB}")
find_package(ruckig 0.19.4 EXACT CONFIG REQUIRED
  PATHS "${CB_PYTHON_PLATLIB}/lib64/cmake/ruckig" NO_DEFAULT_PATH)
ament_python_install_package(${PROJECT_NAME})
ament_get_python_install_dir(CB_PYTHON_INSTALL_DIR)
add_library(companionbot_follow_native SHARED src/motion_authority.cpp src/lightweight_producer.cpp src/follow_algorithms.cpp src/full_reference.cpp)
target_compile_features(companionbot_follow_native PUBLIC cxx_std_17)
target_include_directories(companionbot_follow_native PUBLIC
  $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
target_link_libraries(companionbot_follow_native PUBLIC ruckig::ruckig)
target_compile_options(companionbot_follow_native PRIVATE -Wall -Wextra -Werror -fno-fast-math -ffp-contract=off)
add_library(companionbot_navigation_ros SHARED src/navigation_ingress.cpp)
target_compile_features(companionbot_navigation_ros PUBLIC cxx_std_17)
target_compile_options(companionbot_navigation_ros PRIVATE -Wall -Wextra -Werror -fno-fast-math -ffp-contract=off)
target_link_libraries(companionbot_navigation_ros companionbot_follow_native nlohmann_json::nlohmann_json)
ament_target_dependencies(companionbot_navigation_ros rclcpp companionbot_interfaces geometry_msgs nav_msgs sensor_msgs tf2_msgs)
pybind11_add_module(_native_runtime NO_EXTRAS src/bindings.cpp src/authority_bindings.cpp src/ingress_bindings.cpp src/follow_bindings.cpp src/full_bindings.cpp)
target_link_libraries(_native_runtime PRIVATE companionbot_follow_native companionbot_navigation_ros)
target_compile_options(_native_runtime PRIVATE -Wall -Wextra -Werror -fno-fast-math -ffp-contract=off)
set_target_properties(_native_runtime PROPERTIES INSTALL_RPATH "$ORIGIN/../../..")
install(TARGETS companionbot_follow_native EXPORT export_companionbot_follow LIBRARY DESTINATION lib)
install(TARGETS companionbot_navigation_ros LIBRARY DESTINATION lib)
install(TARGETS _native_runtime LIBRARY DESTINATION "${CB_PYTHON_INSTALL_DIR}/${PROJECT_NAME}")
install(DIRECTORY include/ DESTINATION include)
install(FILES README.md DESTINATION share/${PROJECT_NAME})
if(BUILD_TESTING)
  enable_testing()
  add_executable(motion_authority_test test/native/motion_authority_test.cpp)
  target_link_libraries(motion_authority_test PRIVATE companionbot_follow_native)
  target_compile_options(motion_authority_test PRIVATE -Wall -Wextra -Werror)
  add_test(NAME motion_authority_native COMMAND motion_authority_test)
  add_executable(follow_controller_test test/native/follow_controller_test.cpp)
  target_link_libraries(follow_controller_test PRIVATE companionbot_follow_native Threads::Threads)
  target_compile_options(follow_controller_test PRIVATE -Wall -Wextra -Werror)
  add_test(NAME native_follow_controller COMMAND follow_controller_test)
  find_package(ament_cmake_pytest REQUIRED)
  ament_add_pytest_test(rolling ${CMAKE_CURRENT_SOURCE_DIR}/../../../tests/test_rolling_reference.py)
  ament_add_pytest_test(governor ${CMAKE_CURRENT_SOURCE_DIR}/../../../tests/test_follow_governor_wait.py)
  ament_add_pytest_test(radial ${CMAKE_CURRENT_SOURCE_DIR}/../../../tests/test_stage6_radial_kf_alignment.py)
  ament_add_pytest_test(native_authority ${CMAKE_CURRENT_SOURCE_DIR}/../../../tests/test_native_motion_authority.py)
  ament_add_pytest_test(native_producer ${CMAKE_CURRENT_SOURCE_DIR}/../../../tests/test_native_reference_producer.py)
  ament_add_pytest_test(native_full_reference ${CMAKE_CURRENT_SOURCE_DIR}/../../../tests/test_native_full_reference.py)
  ament_add_pytest_test(native_follow_algorithms ${CMAKE_CURRENT_SOURCE_DIR}/../../../tests/test_native_follow_algorithms.py)
  ament_add_pytest_test(native_follow_authority ${CMAKE_CURRENT_SOURCE_DIR}/../../../tests/test_native_follow_authority.py)
  ament_add_pytest_test(native_follow_controller_parity ${CMAKE_CURRENT_SOURCE_DIR}/../../../tests/test_native_follow_controller.py)
endif()
ament_export_targets(export_companionbot_follow HAS_LIBRARY_TARGET)
ament_export_include_directories(include)
ament_package()
