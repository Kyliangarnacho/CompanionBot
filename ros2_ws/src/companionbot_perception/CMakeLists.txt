cmake_minimum_required(VERSION 3.20)
project(companionbot_perception LANGUAGES CXX)
if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
  set(CMAKE_BUILD_TYPE Release CACHE STRING "Native image processing build profile" FORCE)
endif()
if(DEFINED PYTHON_EXECUTABLE)
  set(Python3_EXECUTABLE "${PYTHON_EXECUTABLE}")
endif()
find_package(ament_cmake REQUIRED)
find_package(ament_cmake_python REQUIRED)
find_package(Python3 REQUIRED COMPONENTS Interpreter Development.Module)
find_package(pybind11 REQUIRED CONFIG)
# The matcher and rectification audit use one installed C++ ABI. The Python
# cv2 wheel is a separate RGB/NumPy client, never a cv::Mat ABI provider.
find_package(OpenCV 4.6.0 EXACT REQUIRED COMPONENTS core imgproc calib3d)
ament_python_install_package(${PROJECT_NAME})
ament_get_python_install_dir(CB_PYTHON_INSTALL_DIR)
add_library(companionbot_stereo_depth SHARED native/src/stereo_depth.cpp)
target_compile_features(companionbot_stereo_depth PUBLIC cxx_std_17)
target_include_directories(companionbot_stereo_depth PUBLIC
  $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/native/include> $<INSTALL_INTERFACE:include>)
target_include_directories(companionbot_stereo_depth SYSTEM PUBLIC ${OpenCV_INCLUDE_DIRS})
target_link_libraries(companionbot_stereo_depth PUBLIC ${OpenCV_LIBS})
target_compile_options(companionbot_stereo_depth PRIVATE -Wall -Wextra -Werror -fno-fast-math -ffp-contract=off)
pybind11_add_module(_native_stereo NO_EXTRAS native/src/bindings.cpp)
target_link_libraries(_native_stereo PRIVATE companionbot_stereo_depth)
target_compile_options(_native_stereo PRIVATE -Wall -Wextra -Werror -fno-fast-math -ffp-contract=off)
set_target_properties(_native_stereo PROPERTIES INSTALL_RPATH "$ORIGIN/../../..")
install(TARGETS companionbot_stereo_depth EXPORT export_companionbot_stereo LIBRARY DESTINATION lib)
install(TARGETS _native_stereo LIBRARY DESTINATION "${CB_PYTHON_INSTALL_DIR}/${PROJECT_NAME}")
install(DIRECTORY native/include/ DESTINATION include)
configure_file(scripts/camera_tf.in camera_tf @ONLY
  FILE_PERMISSIONS OWNER_READ OWNER_WRITE OWNER_EXECUTE GROUP_READ GROUP_EXECUTE WORLD_READ WORLD_EXECUTE)
install(PROGRAMS ${CMAKE_CURRENT_BINARY_DIR}/camera_tf DESTINATION lib/${PROJECT_NAME})
install(FILES README.md DESTINATION share/${PROJECT_NAME})
if(BUILD_TESTING)
  enable_testing()
  find_package(ament_cmake_pytest REQUIRED)
  add_executable(stereo_depth_test native/test/stereo_depth_test.cpp)
  target_link_libraries(stereo_depth_test PRIVATE companionbot_stereo_depth)
  target_compile_options(stereo_depth_test PRIVATE -Wall -Wextra -Werror -UNDEBUG)
  add_test(NAME stereo_depth_native COMMAND stereo_depth_test)
  ament_add_pytest_test(perception_python test)
  # Preserve the original ament_python setup.cfg Stage7 coverage after moving
  # the package build owner to ament_cmake for the optional native channel.
  foreach(case demo_yolo26n_master_lock perception_camera perception_depth
      perception_detection_stream perception_master_selection perception_person_detector
      perception_person_reid perception_person_tracker stage7_6_reid stage7_7_depth
      stage7_7_master_depth stage7_final_session stage7_robot_geometry)
    ament_add_pytest_test(${case} ${CMAKE_CURRENT_SOURCE_DIR}/../../../tests/test_${case}.py)
  endforeach()
  ament_add_pytest_test(native_stereo ${CMAKE_CURRENT_SOURCE_DIR}/../../../tests/test_native_stereo.py)
  ament_add_pytest_test(stereo_runtime_contract ${CMAKE_CURRENT_SOURCE_DIR}/../../../tests/test_stereo_runtime_contract.py)
  ament_add_pytest_test(rendered_perception_contract ${CMAKE_CURRENT_SOURCE_DIR}/../../../tests/test_rendered_perception_contract.py)
  ament_add_pytest_test(rendered_thread_budget ${CMAKE_CURRENT_SOURCE_DIR}/../../../tests/test_rendered_thread_budget.py)
endif()
ament_export_targets(export_companionbot_stereo HAS_LIBRARY_TARGET)
ament_export_include_directories(include)
ament_export_dependencies(OpenCV)
ament_package()
