cmake_minimum_required(VERSION 3.20)
project(companionbot_simulation LANGUAGES CXX)
if(DEFINED PYTHON_EXECUTABLE)
  set(Python3_EXECUTABLE "${PYTHON_EXECUTABLE}")
endif()
find_package(ament_cmake REQUIRED)
find_package(ament_cmake_python REQUIRED)
find_package(Python3 REQUIRED COMPONENTS Interpreter Development.Module)
find_package(pybind11 REQUIRED CONFIG)
execute_process(COMMAND "${Python3_EXECUTABLE}" -c
  "import sysconfig; print(sysconfig.get_path('platlib'))"
  OUTPUT_VARIABLE CB_PYTHON_PLATLIB OUTPUT_STRIP_TRAILING_WHITESPACE
  COMMAND_ERROR_IS_FATAL ANY)
find_package(ruckig 0.19.4 EXACT CONFIG REQUIRED
  PATHS "${CB_PYTHON_PLATLIB}/lib64/cmake/ruckig" NO_DEFAULT_PATH)
find_package(companionbot_follow REQUIRED)
find_package(companionbot_control REQUIRED)
find_package(nlohmann_json REQUIRED)
ament_python_install_package(${PROJECT_NAME})
ament_get_python_install_dir(CB_PYTHON_INSTALL_DIR)
add_library(companionbot_navigation_sim SHARED native/src/navigation_runtime.cpp)
target_compile_features(companionbot_navigation_sim PUBLIC cxx_std_17)
target_include_directories(companionbot_navigation_sim PUBLIC
  $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/native/include> $<INSTALL_INTERFACE:include>)
target_link_libraries(companionbot_navigation_sim PUBLIC
  companionbot_follow::companionbot_follow_native companionbot_control::companionbot_core nlohmann_json::nlohmann_json)
target_compile_options(companionbot_navigation_sim PRIVATE -Wall -Wextra -Werror -fno-fast-math -ffp-contract=off)
pybind11_add_module(_native_simulation NO_EXTRAS native/src/bindings.cpp)
target_link_libraries(_native_simulation PRIVATE companionbot_navigation_sim)
target_compile_options(_native_simulation PRIVATE -Wall -Wextra -Werror -fno-fast-math -ffp-contract=off)
set_target_properties(_native_simulation PROPERTIES INSTALL_RPATH "$ORIGIN/../../..")
install(TARGETS companionbot_navigation_sim LIBRARY DESTINATION lib)
install(TARGETS _native_simulation LIBRARY DESTINATION "${CB_PYTHON_INSTALL_DIR}/${PROJECT_NAME}")
foreach(entry frozen_sim spatial_sim)
  configure_file(scripts/${entry}.in ${entry} @ONLY
    FILE_PERMISSIONS OWNER_READ OWNER_WRITE OWNER_EXECUTE GROUP_READ GROUP_EXECUTE WORLD_READ WORLD_EXECUTE)
  install(PROGRAMS ${CMAKE_CURRENT_BINARY_DIR}/${entry} DESTINATION lib/${PROJECT_NAME})
endforeach()
install(FILES README.md DESTINATION share/${PROJECT_NAME})
if(BUILD_TESTING)
  enable_testing()
  find_package(ament_cmake_pytest REQUIRED)
  add_executable(navigation_runtime_test native/test/navigation_runtime_test.cpp)
  target_link_libraries(navigation_runtime_test PRIVATE companionbot_navigation_sim)
  target_compile_options(navigation_runtime_test PRIVATE -Wall -Wextra -Werror -UNDEBUG)
  add_test(NAME navigation_runtime_native COMMAND navigation_runtime_test)
  ament_add_pytest_test(simulation_python test)
  ament_add_pytest_test(packet_safety ${CMAKE_CURRENT_SOURCE_DIR}/../../../tests/test_stage6_packet_safety.py)
  ament_add_pytest_test(native_navigation_runtime ${CMAKE_CURRENT_SOURCE_DIR}/../../../tests/test_native_navigation_runtime.py)
endif()
ament_package()
