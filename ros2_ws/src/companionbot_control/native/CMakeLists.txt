cmake_minimum_required(VERSION 3.16)
project(companionbot_core C)
option(CB_BUILD_TESTS "Build host control tests" ON)
add_library(companionbot_core src/control.c src/estimator.c src/mcu.c src/wire.c)
target_include_directories(companionbot_core PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
target_compile_features(companionbot_core PUBLIC c_std_99)
target_compile_options(companionbot_core PRIVATE -Wall -Wextra -Werror -Wpedantic -fno-fast-math -ffp-contract=off)
target_link_libraries(companionbot_core PUBLIC m)
set_target_properties(companionbot_core PROPERTIES POSITION_INDEPENDENT_CODE ON)
if(CB_BUILD_TESTS AND NOT CMAKE_CROSSCOMPILING)
  enable_testing()
  add_executable(control_core_test test/core_test.c)
  target_link_libraries(control_core_test PRIVATE companionbot_core)
  target_compile_options(control_core_test PRIVATE -Wall -Wextra -Werror -UNDEBUG)
  add_test(NAME control_core COMMAND control_core_test)
  add_executable(mcu_safety_test test/safety_test.c)
  target_link_libraries(mcu_safety_test PRIVATE companionbot_core)
  target_compile_options(mcu_safety_test PRIVATE -Wall -Wextra -Werror -UNDEBUG)
  foreach(case normal_barrier armed_cancel braking_revoke duplicate_stop lower_authority new_authorization heartbeat_revoke)
    add_test(NAME mcu_safety_${case} COMMAND mcu_safety_test ${case})
  endforeach()
endif()
