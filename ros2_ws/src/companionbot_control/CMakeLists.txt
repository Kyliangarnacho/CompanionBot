cmake_minimum_required(VERSION 3.20)
project(companionbot_control LANGUAGES C CXX)
if(DEFINED PYTHON_EXECUTABLE)
  set(Python3_EXECUTABLE "${PYTHON_EXECUTABLE}")
endif()
find_package(ament_cmake REQUIRED)
find_package(ament_cmake_python REQUIRED)
find_package(rclcpp REQUIRED)
find_package(companionbot_interfaces REQUIRED)
find_package(diagnostic_msgs REQUIRED)
find_package(tf2_msgs REQUIRED)
find_package(ament_index_cpp REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(tf2_ros REQUIRED)
find_package(rcl_interfaces REQUIRED)
find_package(nlohmann_json REQUIRED)
find_package(OpenSSL REQUIRED)
find_package(Threads REQUIRED)
set(BUILD_SHARED_LIBS ON)
set(CB_BUILD_TESTS ${BUILD_TESTING} CACHE BOOL "Build native tests")
add_subdirectory(native)
ament_python_install_package(${PROJECT_NAME})
add_library(companionbot_spatial src/spatial_state.cpp)
target_compile_features(companionbot_spatial PUBLIC cxx_std_17)
target_compile_options(companionbot_spatial PRIVATE -Wall -Wextra -Werror -fno-fast-math -ffp-contract=off)
target_include_directories(companionbot_spatial PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
target_link_libraries(companionbot_spatial PUBLIC OpenSSL::Crypto nlohmann_json::nlohmann_json)
add_executable(spatial_state src/spatial_state_node.cpp)
target_compile_features(spatial_state PRIVATE cxx_std_17)
target_compile_options(spatial_state PRIVATE -Wall -Wextra -Werror -fno-fast-math -ffp-contract=off)
target_link_libraries(spatial_state companionbot_spatial)
ament_target_dependencies(spatial_state rclcpp ament_index_cpp companionbot_interfaces
  diagnostic_msgs geometry_msgs nav_msgs sensor_msgs tf2_ros rcl_interfaces)
install(TARGETS companionbot_spatial LIBRARY DESTINATION lib)
install(TARGETS spatial_state DESTINATION lib/${PROJECT_NAME})
add_library(companionbot_transport src/serial_port.cpp)
target_compile_features(companionbot_transport PUBLIC cxx_std_17)
target_compile_options(companionbot_transport PRIVATE -Wall -Wextra -Werror)
target_include_directories(companionbot_transport PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
add_executable(mcu_bridge src/mcu_bridge.cpp)
target_compile_features(mcu_bridge PRIVATE cxx_std_17)
target_compile_options(mcu_bridge PRIVATE -Wall -Wextra -Werror)
target_link_libraries(mcu_bridge companionbot_transport companionbot_core Threads::Threads)
ament_target_dependencies(mcu_bridge rclcpp companionbot_interfaces diagnostic_msgs)
install(TARGETS companionbot_core companionbot_transport EXPORT export_companionbot_control LIBRARY DESTINATION lib ARCHIVE DESTINATION lib)
install(TARGETS mcu_bridge DESTINATION lib/${PROJECT_NAME})
add_executable(tf_authority src/tf_authority.cpp)
target_compile_features(tf_authority PRIVATE cxx_std_17)
target_compile_options(tf_authority PRIVATE -Wall -Wextra -Werror)
ament_target_dependencies(tf_authority rclcpp diagnostic_msgs tf2_msgs)
install(TARGETS tf_authority DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY native/include/ include/ DESTINATION include)
install(FILES README.md DESTINATION share/${PROJECT_NAME})
if(BUILD_TESTING)
  add_executable(spatial_native_test test/spatial_native_test.cpp)
  target_link_libraries(spatial_native_test companionbot_spatial)
  target_compile_options(spatial_native_test PRIVATE -Wall -Wextra -Werror)
  add_test(NAME spatial_native COMMAND spatial_native_test)
  add_executable(spatial_replay test/spatial_replay.cpp)
  target_link_libraries(spatial_replay companionbot_spatial)
  target_compile_options(spatial_replay PRIVATE -Wall -Wextra -Werror)
  find_package(ament_cmake_pytest REQUIRED)
  ament_add_pytest_test(control_python test)
  ament_add_pytest_test(slope ${CMAKE_CURRENT_SOURCE_DIR}/../../../tests/test_slope_estimation.py)
  ament_add_pytest_test(slip ${CMAKE_CURRENT_SOURCE_DIR}/../../../tests/test_slip_estimation.py)
  ament_add_pytest_test(runtime ${CMAKE_CURRENT_SOURCE_DIR}/../../../tests/test_stage4_runtime.py)
  ament_add_pytest_test(authority ${CMAKE_CURRENT_SOURCE_DIR}/../../../tests/test_stage6_2d_split_authority.py)
  ament_add_pytest_test(embedded ${CMAKE_CURRENT_SOURCE_DIR}/../../../tests/test_embedded_native.py)
  ament_add_pytest_test(spatial ${CMAKE_CURRENT_SOURCE_DIR}/../../../tests/test_spatial_state.py)
  ament_add_pytest_test(native_spatial ${CMAKE_CURRENT_SOURCE_DIR}/../../../tests/test_native_spatial.py)
endif()
ament_export_targets(export_companionbot_control HAS_LIBRARY_TARGET)
ament_export_include_directories(include)
ament_package()
